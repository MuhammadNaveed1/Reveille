  var ALT_MIN=-9, ALT_MAX=12;

  // ---------------------------------------------------------
  // SOLAR POSITION (real, ~0.5-1 deg accuracy)
  // ---------------------------------------------------------
  function rad(d){return d*Math.PI/180;} function deg(r){return r*180/Math.PI;}
  function julianDate(d){return d.getTime()/86400000+2440587.5;}
  function sunGeom(date,lat,lng){
    var n=julianDate(date)-2451545.0;
    var L=(280.460+0.9856474*n)%360; if(L<0)L+=360;
    var g=(357.528+0.9856003*n)%360; if(g<0)g+=360;
    var lambda=L+1.915*Math.sin(rad(g))+0.020*Math.sin(rad(2*g)), eps=23.439-0.0000004*n;
    var ra=deg(Math.atan2(Math.cos(rad(eps))*Math.sin(rad(lambda)),Math.cos(rad(lambda))));
    var dec=deg(Math.asin(Math.sin(rad(eps))*Math.sin(rad(lambda))));
    var GMST=(280.46061837+360.98564736629*n)%360; if(GMST<0)GMST+=360;
    var LST=(GMST+lng)%360; if(LST<0)LST+=360;
    var HA=LST-ra; if(HA>180)HA-=360; if(HA<-180)HA+=360;
    var alt=deg(Math.asin(Math.sin(rad(lat))*Math.sin(rad(dec))+Math.cos(rad(lat))*Math.cos(rad(dec))*Math.cos(rad(HA))));
    return { alt:alt, rising: HA<0 };
  }

  // ---------------------------------------------------------
  // OKLCH -> sRGB
  // ---------------------------------------------------------
  function clamp(v,a,b){return Math.max(a,Math.min(b,v));}
  function lerp(a,b,t){return a+(b-a)*t;}
  function smooth(t){return t*t*(3-2*t);}
  function lerpHue(a,b,t){ var d=((b-a+540)%360)-180; return (a+d*t+360)%360; }
  function oklchToRgb(Lin,Cin,Hd){
    var h=Hd*Math.PI/180, a=Cin*Math.cos(h), b=Cin*Math.sin(h);
    var l_=Lin+0.3963377774*a+0.2158037573*b, m_=Lin-0.1055613458*a-0.0638541728*b, s_=Lin-0.0894841775*a-1.2914855480*b;
    l_=l_*l_*l_; m_=m_*m_*m_; s_=s_*s_*s_;
    var r=4.0767416621*l_-3.3077115913*m_+0.2309699292*s_;
    var g=-1.2684380046*l_+2.6097574011*m_-0.3413193965*s_;
    var bb=-0.0041960863*l_-0.7034186147*m_+1.7076147010*s_;
    function toS(c){ c=clamp(c,0,1); return c<=0.0031308?12.92*c:1.055*Math.pow(c,1/2.4)-0.055; }
    return [Math.round(toS(r)*255),Math.round(toS(g)*255),Math.round(toS(bb)*255)];
  }
  function hex(rgb){ return '#'+rgb.map(function(v){return clamp(v,0,255).toString(16).padStart(2,'0');}).join(''); }

  // ---------------------------------------------------------
  // STRATA GRADIENT ENGINE (deterministic anchors from sun+weather)
  // ---------------------------------------------------------
  function assembleAnchors(alt, atm){
    var c01=atm.cloud/100, h01=atm.humidity/100, a01=Math.min(1,atm.aqi/200);
    var haze=clamp(1-atm.visibility/40,0,1);
    var altT=clamp((alt-ALT_MIN)/(ALT_MAX-ALT_MIN),0,1);
    var brt=lerp(0,0.16,smooth(altT));
    var sat=(1-0.30*h01)*(1-0.35*haze);
    var cloudMul=c01<=0.6?1+0.35*Math.max(0,1-Math.abs(c01-0.30)/0.30):1-0.55*((c01-0.6)/0.4);
    cloudMul=clamp(cloudMul,0.25,1.4);
    var warmShift=a01*9+haze*6, horizDark=a01*0.10, horizLift=haze*0.10;
    var overcast=atm.cloud>70, heavy=atm.cloud>85, clearish=atm.cloud<55,
        lowSun=alt<4, smoggy=atm.aqi>90, hazy=(atm.visibility<16||atm.humidity>72);
    var A=[];
    function add(pos,L,C,H){ A.push({pos:pos,L:clamp(L,0.05,0.97),C:clamp(C*sat,0,0.30),H:((H%360)+360)%360}); }
    add(0.00, 0.46+brt*1.3-(overcast?0.03:0), overcast?0.025:0.085, overcast?250:256);
    if(!overcast) add(0.34, 0.60+brt, 0.075, 262);
    if(lowSun && clearish && !smoggy) add(0.50, 0.66+brt*0.7, 0.045, 286);
    if(alt>-6 && alt<6 && !overcast && (atm.aqi>20 || (atm.cloud>=10 && atm.cloud<=60)))
      add(0.63, 0.76+brt*0.6, 0.085*cloudMul*(1+0.40*a01), 338);
    if(!heavy) add(0.78, 0.74+brt*0.5+horizLift-horizDark, 0.110*cloudMul*(1+0.20*a01), 32-warmShift);
    if(smoggy || atm.visibility<12) add(0.88, 0.66+horizLift-horizDark*1.4, 0.070*(1+0.30*a01), 26-warmShift);
    if(hazy) add(0.93, 0.72+brt*0.3+horizLift, 0.050*(1-0.45*haze), 64-warmShift*0.6);
    add(1.00, overcast?0.60+brt*0.2:0.72+brt*0.3+horizLift-horizDark, overcast?0.030:(hazy?0.045:0.070), overcast?72:70-warmShift*0.6);
    var nightF=clamp((ALT_MIN-alt)/21,0,1);
    if(nightF>0){ A.forEach(function(q){ q.L=lerp(q.L,q.L*0.20,nightF); q.C*=1-0.65*nightF; q.H=lerpHue(q.H,250,0.5*nightF); }); }
    return A;
  }
  function buildGradientCSS(alt, atm, N){
    var cps=assembleAnchors(alt,atm), out=[];
    for(var i=0;i<N;i++){
      var t=N===1?0:i/(N-1), s=0;
      while(s<cps.length-2 && t>cps[s+1].pos) s++;
      var a=cps[s], b=cps[s+1];
      var f=b.pos===a.pos?0:smooth(clamp((t-a.pos)/(b.pos-a.pos),0,1));
      var L=lerp(a.L,b.L,f), C=lerp(a.C,b.C,f), H=lerpHue(a.H,b.H,f);
      out.push({pos:t*100, color:hex(oklchToRgb(L,C,H))});
    }
    return 'linear-gradient(to bottom,'+out.map(function(s){return s.color+' '+s.pos.toFixed(2)+'%';}).join(',')+')';
  }

  // WEATHER — Open-Meteo live, seeded mock fallback

  function hashStr(str){ var h=2166136261; for(var i=0;i<str.length;i++){h^=str.charCodeAt(i);h=Math.imul(h,16777619);} return (h>>>0)/4294967295; }
  function mockWeather(c){ function r(k){return hashStr(c.name+'|'+k);} var p=c.profile; return {
    cloud:clamp(Math.round(p.cloud+(r('c')*2-1)*32),0,100),
    humidity:clamp(Math.round(p.humidity+(r('h')*2-1)*20),0,100),
    visibility:clamp(Math.round(p.visibility+(r('v')*2-1)*14),1,50),
    aqi:clamp(Math.round(p.aqi+(r('a')*2-1)*45),0,300) }; }

  var weatherCache={ data:{}, live:false };
  function fetchWeatherAll(){
    var lat=CITIES.map(function(c){return c.lat;}).join(',');
    var lng=CITIES.map(function(c){return c.lng;}).join(',');
    Promise.all([
      fetch('https://api.open-meteo.com/v1/forecast?latitude='+lat+'&longitude='+lng+'&current=cloud_cover_high,relative_humidity_2m,visibility').then(function(r){return r.json();}),
      fetch('https://air-quality-api.open-meteo.com/v1/air-quality?latitude='+lat+'&longitude='+lng+'&current=us_aqi').then(function(r){return r.json();})
    ]).then(function(res){
      var F=Array.isArray(res[0])?res[0]:[res[0]], Aq=Array.isArray(res[1])?res[1]:[res[1]];
      var data={};
      CITIES.forEach(function(c,i){
        data[c.name]={
          cloud: (F[i]&&F[i].current&&F[i].current.cloud_cover_high!=null)?F[i].current.cloud_cover_high:c.profile.cloud,
          humidity: (F[i]&&F[i].current&&F[i].current.relative_humidity_2m!=null)?F[i].current.relative_humidity_2m:c.profile.humidity,
          visibility: Math.round(((F[i]&&F[i].current&&F[i].current.visibility!=null)?F[i].current.visibility:c.profile.visibility*1000)/1000),
          aqi: (Aq[i]&&Aq[i].current&&Aq[i].current.us_aqi!=null)?Aq[i].current.us_aqi:c.profile.aqi
        };
      });
      weatherCache={ data:data, live:true };
      paint();
    }).catch(function(e){ console.warn('Reveille: live weather failed, using mock', e); weatherCache.live=false; paint(); });
  }
  function getWeather(city){
    if(weatherCache.live && weatherCache.data[city.name]) return weatherCache.data[city.name];
    return mockWeather(city);
  }

  // ---------------------------------------------------------
  // FOCAL CITY — closest to peak dawn (alt ~0.5deg, rising).
  // Honolulu sits in CITIES too, so it naturally becomes focal
  // during the pacific gap when nobody else is near dawn.
  // ---------------------------------------------------------
  function pickFocalCity(date){
    var best=null, bestScore=Infinity;
    CITIES.forEach(function(c){
      var g=sunGeom(date,c.lat,c.lng);
      var score = g.rising ? Math.abs(g.alt-0.5) : 200+Math.abs(g.alt-0.5);
      if(score<bestScore){ bestScore=score; best={city:c, alt:g.alt}; }
    });
    return best;
  }

  function tzTimeString(date, tz){
    try{
      return date.toLocaleTimeString('en-US',{timeZone:tz,hour:'numeric',minute:'2-digit',hour12:true}).toLowerCase();
    }catch(e){ return '—'; }
  }

  // ---------------------------------------------------------
  // PAINT — one calculation, two targets (loader sky + hero bg)
  // ---------------------------------------------------------
  var heroBgEl = null;
  var loaderSkyEl = null;
  var loaderActive = true;

  function paint(){
    var date = new Date();
    var focal = pickFocalCity(date);
    var atm = getWeather(focal.city);
    var css = buildGradientCSS(focal.alt, atm, 48);

    if(loaderActive && loaderSkyEl) loaderSkyEl.style.background = css;
    if(heroBgEl) heroBgEl.style.background = css;

    if(loaderActive && SHOW_LOADER_TEXT){
      var label = focal.city.name==='Honolulu' ? 'Standing Watch' : 'First Light';
      var cityEl = document.getElementById('reveille-loader-city');
      var timeEl = document.getElementById('reveille-loader-time');
      var labelEl = document.getElementById('reveille-loader-label');
      if(labelEl) labelEl.textContent = label;
      if(cityEl) cityEl.textContent = focal.city.name;
      if(timeEl) timeEl.textContent = tzTimeString(date, focal.city.tz);
    }
  }

  // ---------------------------------------------------------
  // LOADER LIFECYCLE
  // ---------------------------------------------------------
  function dismissLoader(){
    var loader = document.getElementById('reveille-loader');
    if(!loader) return;
    loader.classList.add('rvl-fade');
    document.documentElement.style.overflow='';
    document.body.style.overflow='';
    setTimeout(function(){
      loaderActive = false;
      if(loader.parentNode) loader.parentNode.removeChild(loader);
    }, LOADER_FADE_MS + 50);
  }

  function init(){
    heroBgEl = document.getElementById(HERO_BG_ID);
    loaderSkyEl = document.getElementById('reveille-loader-sky');

    // lock scroll while loader is up
    document.documentElement.style.overflow='hidden';
    document.body.style.overflow='hidden';

    paint(); // first paint — instant, no flash on either target

    if(SHOW_LOADER_TEXT){
      setTimeout(function(){
        var meta=document.getElementById('reveille-loader-meta');
        if(meta) meta.classList.add('rvl-ready');
      }, 120);
    }

    setTimeout(dismissLoader, INTRO_REVEAL_MS + LOADER_MIN_HOLD_MS);

    fetchWeatherAll();
    setInterval(fetchWeatherAll, WEATHER_REFRESH_MS);
    setInterval(paint, BG_TICK_MS); // keeps hero-bg + (while up) loader in sync going forward
  }

  if(document.readyState==='loading'){
    document.addEventListener('DOMContentLoaded', init);
  } else {
    init();
  }
})();
